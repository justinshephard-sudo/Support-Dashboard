// NOTE 2026-11-10: the live workflow (01a18341) also handles kind "signals" (AI check of Call signals); get_workflow has the current source.
import { defineDurable } from "@zapier/zapier-durable";
import { createZapierSdk } from "@zapier/zapier-sdk";
import { z } from "zod";
import { createHash } from "node:crypto";

// CS Dashboard "Call prep": the call-prep Apps Script sends the prompt, the firm context and the
// last Gong transcript; Claude (through AI by Zapier) writes the prep sheet as JSON; this posts
// it back to the Apps Script, which hands it to the dashboard pop-up.
const sdk = createZapierSdk();

// Only the call-prep Apps Script can start a run: it sends hookKey (kept in its git-ignored secret.js), and
// this holds just the key's SHA-256. Results only ever go back to that script's web app, never to a
// callbackUrl from the request, so the public hook can't be used to run Claude for someone else.
const HOOK_KEY_SHA256 = "5c81bc12f2cc73346dc1d4ac188bea5b7b6bcb803f8e9dfcf5054341b26232f7";
const CALLBACK_URL = "https://script.google.com/macros/s/AKfycbwltwhfY4-dLO6F5uyfzxIXIMcb_N0_zrk-2q0atrk_c8fMj4USTTcEJK4QsChbmAVR/exec";

const InputSchema = z.object({
  requestId: z.string(),
  token: z.string(),
  hookKey: z.string(),
  instructions: z.string(),
  context: z.string().nullable().optional(),
  transcript: z.string().nullable().optional(),
}).loose();
type Input = z.infer<typeof InputSchema>;

// The trigger is a raw catch hook (Zapier's parsed hook once dropped a long transcript's body), so
// the request body arrives as a string under content/body; parse it here.
function normalizeInput(raw: unknown): unknown {
  try {
    if (typeof raw === "string") return JSON.parse(raw);
    const r = (raw ?? {}) as Record<string, unknown>;
    if (typeof r.requestId === "string") return r;
    const body = [r.content, r.body, r.raw_body].find((v) => typeof v === "string" && v.trim().startsWith("{"));
    return body ? JSON.parse(body as string) : r;
  } catch {
    return {};
  }
}

// The model wraps JSON in a code fence now and then; take the outermost object.
function parsePrep(text: string): unknown {
  const s = text.indexOf("{"), e = text.lastIndexOf("}");
  if (s < 0 || e <= s) return null;
  try { return JSON.parse(text.slice(s, e + 1)); } catch { return null; }
}

export default defineDurable<Input, unknown>("cs-dashboard-call-prep", async (ctx, rawInput) => {
  // The hook URL also gets empty pings (no body) from outside checks; skip those quietly instead of failing.
  const parsed = InputSchema.safeParse(normalizeInput(rawInput));
  if (!parsed.success) return { skipped: true, reason: "not a call-prep request" };
  const input = parsed.data;
  if (createHash("sha256").update(input.hookKey).digest("hex") !== HOOK_KEY_SHA256) return { skipped: true, reason: "bad key" };

  const completion = await ctx.step("write-call-prep", async () =>
    sdk.runAction({
      appKey: "AICLIAPI",
      actionType: "write",
      actionKey: "get_completion",
      inputs: {
        provider_id: "anthropic",
        authentication_id: "0",
        model_id: "anthropic/claude-sonnet-5-5",
        instructions: input.instructions,
        inputFields: {
          firm: input.context ?? "",
          transcript: input.transcript ?? "(No Gong recording with this firm yet.)",
        },
        includeWorkflowData: false,
        outputFields: [{ name: "prep_json", type: "text", isRequired: true }],
      },
    }),
  );

  const item = (completion.data[0] ?? {}) as Record<string, unknown>;
  const text = String(item.prep_json ?? item.output ?? "");
  const prep = parsePrep(text);

  const payload = JSON.stringify({
    action: "prepResult", requestId: input.requestId, token: input.token,
    prep, error: prep ? null : "The AI's answer couldn't be read. Try Regenerate.",
  });

  // Workflow code can't reach outside hosts directly, so the result goes back through Webhooks by Zapier.
  const delivered = await ctx.step("send-prep-to-dashboard", async () =>
    sdk.runAction({
      appKey: "WebHookCLIAPI",
      actionType: "write",
      actionKey: "custom",
      inputs: {
        url: CALLBACK_URL,
        method: "POST",
        data: payload,
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        unflatten: "no",
      },
    }),
  );

  return { requestId: input.requestId, parsed: !!prep, delivered: delivered.data[0] ?? null };
});
